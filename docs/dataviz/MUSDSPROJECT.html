# creating function to make it so that all datasets can be added together cleanly
standardize_data <- function(filename) {
  # take the year from the name of each file, store for later
  year <- str_sub(filename, 1, 4) 
  print(filename)
  
  # select one specific file each round 
  cleaning_data <- all_years_data[[filename]]
    
    # cleaning each column up
    cleaning_data <- cleaning_data |>
      mutate(
        
      # standardize date column
        Date = 
          case_when(
            # if its already in the right format, run mdy
            !is.na(mdy(Date)) ~ 
              mdy(Date, quiet = FALSE),
            
            str_detect(Date, "^\\d{1,2}-[A-Za-z]+$") ~
              paste0(Date, "-", year) |>
              dmy(quiet = FALSE),
            # if its not,then remove the period, replace the slashes with spaces, and put the year a the end
            is.na(mdy(Date)) ~  
              str_remove(Date, "^(Sun|Mon|Tue|Wed|Thu|Fri|Sat)+\\.?\\s+") |>
              str_replace("Aprl|Ap\\.", "April") |>
              str_remove_all( "\\.") |>
              str_replace_all("/", paste0(" ")) |>
              str_replace("$", paste0(" ", year)) |>
              mdy(quiet = FALSE)
              ),
          
      # standardize count column
        Count = case_when(
          
        # creating conditions for count values to be converted
          
          # if its #s keep it that way as number value
          str_detect(Count, "^\\d+$") ~ as.numeric(Count),
          # finds average of ranges, such as 75-82
          str_detect(Count, "^\\d+-\\d+$|^\\d+/\\d+$") ~
          (as.numeric(str_extract(Count, "^\\d+")) +
             as.numeric(str_extract(Count, "\\d+$"))) / 2,
          # for approximations (~) or question marks -- (removes symbol)
          str_detect(Count, "^~\\d+$|^\\d+\\?$") ~ 
            as.numeric(str_remove(Count, "~|\\?")),
          str_detect(Count, "^\\d+~$") ~
            as.numeric(str_remove(Count, "~")),
          # numbers with +, such as 176+ or 600+ FULL
          str_detect(Count, "^\\d+\\s*\\+") ~
            as.numeric(str_extract(Count, "^\\d+")),
          # estimated numbers, such as 131 (est)
          str_detect(Count, "^\\d+\\s*\\(est\\)") ~ 
            as.numeric(str_extract(Count, "^\\d+")),
          # removes abbreviations, such as c.279 or c279
          str_detect(Count, "^c\\.\\s*\\d+$|^c\\d+$|(c\\.\\d+)$") ~
            as.numeric(str_extract(Count, "\\d+")),
          # removes the word full, such as 244 - full or -plus
          str_detect(Count, "^\\d+\\s*-\\s*full$|^\\d+-plus$|^\\d+\\s*-\\s*looked") ~
            as.numeric(str_extract(Count, "^\\d+")),
  
        # if none of the conditions above are met, turn it to NA
        
          # everything else becomes NA (blanks, ?, cancelled, etc.)
          TRUE ~ NA_real_),
      
      
      # standardize location column
      Location = case_when(
        str_detect(Location,"Bridges Hall of Music|Bridges Hall of M|Bridges Hall of Music front patio|Little Bridges") ~
          "Little Bridges",
        str_detect(Location,"Bridges Auditorium") ~
          "Big Bridges",
        str_detect(Location,"Garrison") ~
          "Garrison Theater",
        str_detect(Location,"Benton Museum-Pomona College") ~
          "Benton Museum",
        str_detect(Location,"Smith CC/Rose Thea\\.|Smith Campus Center") ~
          "Smith Campus Center",
        str_detect(Location,"Bryant") ~
          "Bryant Hall",
        str_detect(Location,"Sontag") ~
          "Sontag Greek Theater",
        !str_detect(Location, "Bridges|Garrison|Benton|Smith|Bryant|Lyman|Sontag") ~
          "Non-5C Location",
        TRUE ~ Location
      ),
      
      
      # create event column
      Event_Name = Event ,
      Event = str_to_lower(Event),
      Event_Type = case_when(
        str_detect(Event, "senior|sr\\.|sr|student|stud\\.|jr\\.|jr|soph|soph\\.|class recital|stud\\.rec\\.") ~
          "Student Project/Recital",
        str_detect(Event, "pco & pcc|orchesta & choir|choir & orchestra|pomona college orchestra|orchestra|pc orch|pco|pomona college choir|choir|pc choir|pcc|pomona college band|band|pc band|west african|afro-cuban|gamelan") ~
          "Student Ensemble",
        str_detect(Event, "ensemble|ens|ens\\.|celliola|cornucopia|mojave|artifex|prism|trio|quartet|duo") ~
          "Other Ensemble Performance",
        str_detect(Event, "peterson|^lee|rockwell|pelev|wright|fac\\.|fnc|friday noon|solo") ~
          "Faculty Recital",
        str_detect(Event, "guest|chirgilchin|millennium consort") ~
          "Guest Artist",
       str_detect(Event, "lecture|demo|demonstration") ~
          "Lecture/Demonstration",
        str_detect(Event, "through the gates|convocation|memorial|baccalaureate") ~
          "Non-Music Ceremony",        
       str_detect(Event, "glee") ~
         "Glee Club",
        TRUE ~ "Other"        
        #str_detect(Event, "dance") ~
         # "Dance Performance",
        
        # add student ensembles (theres 8 of them)
        #str_detect(Event, "pco & pcc|orchesta & choir|choir & orchestra") ~
         # "Pomona College Orchestra & Choir",
       # str_detect(Event, "pomona college orchestra|orchestra|pc orch|pco") ~
        #  "Pomona College Orchestra",
       # str_detect(Event, "pomona college band|band|pc band") ~
          #"Pomona College Band",
       # str_detect(Event, "pomona college choir|choir|pc choir|pcc") ~
          #"Pomona College Choir",
       # add quartets/trios/chorale/non-student ensembles

      ),
      
      # create day of week column 
      Day_of_Week = weekdays(Date),
      Day_of_Week = factor(Day_of_Week,
                           levels = c("Monday", "Tuesday", "Wednesday", "Thursday",
               "Friday", "Saturday", "Sunday")),
      Genre = case_when(
        str_detect(Event, "mozart|beethoven|classical|bach|baroque|chamber") ~ 
          "Classical",
        str_detect(Event, "afro|african|brasil|korea|korean|middle east|china|chinese|punjabi|mariachi") ~
          "International",
        str_detect(Event, "jazz") ~
          "Jazz",
        str_detect(Event, "electronic|ussachevsky") ~
          "Electronic",
        TRUE ~ "Other" 
      ),
      
      Instrument = case_when(
        str_detect(Event, "piano|organ|lee|keys|pianist|peterson") ~
          "Keys",
        str_detect(Event, "drum|drums|percussion|perc\\.|perc") ~
          "Percussion",
        str_detect(Event, "guitar|bass|strings|string|celliola|cello|viola|cord a vide|corde a vide|harp|wright") ~
          "Strings",
        str_detect(Event, "horn|horns|brass") ~
          "Brass",
        str_detect(Event, "pco & pcc|orchesta & choir|choir & orchestra|pco|pomona college orchestra|orchestra|pc orch|pomona college band|band|pc band|gamelan|jazz|balinese|afro cuban|afro-cuban|mariachi|merengue") ~
          "Multi",
        str_detect(Event, "vocal|singer|singing|sing|voice|acapella|a capella|glee|choir|opera|lytle|givens") ~
          "Voice",
        str_detect(Event, "electronic|ussachevsky") ~
          "Electronic",
        
        TRUE ~ "Other"
      ),
      
      Semester_Year = semester(Date, with_year = TRUE)

      )
        
          
    
    # check for time column
    if ("Time" %in% names(cleaning_data)) {
      
      if (inherits(cleaning_data$Time, "difftime")) {
        cleaning_data <- cleaning_data |>
          mutate(Time2 = hms::as_hms(Time))
      } else {
      cleaning_data <- cleaning_data |>
        
        # standardize time values
        mutate(Time2 = parse_date_time(
          Time,
          # acceptable formats: 12hr:MinSecpm, 12hrpm, 12hr pm, 12hr:Minpm
          orders = c("I:M p", "I p", "Ip", "I:Mp"),
          quiet = FALSE)
    
        ) |>
        mutate(Time2 = hms::as_hms(Time2))
      
      }
          
    
  # return clean/updated dataset of each semester
    # make original Time column the same type across all files
  cleaning_data <- cleaning_data |>
    mutate(Time = as.character(Time))

    }
    
    
    cleaning_data
}
